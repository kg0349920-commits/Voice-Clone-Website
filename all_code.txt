    let selectedFile = null;

    /*
        Show selected voice file.
    */
    document.getElementById("voiceFile").addEventListener("change", function () {

        selectedFile = this.files[0];

        let fileName = document.getElementById("fileName");

        if (selectedFile) {

            fileName.style.display = "block";

            fileName.innerText =
                "Selected: " + selectedFile.name;
        }

    });


    /*
        Show a message to the user.
    */
    function showMessage(text, type) {

        let message = document.getElementById("message");

        message.style.display = "block";

        message.className = "message " + type;

        message.innerText = text;
    }


    /*
        Generate cloned speech.
    */
    async function generateVoice() {

        let text = document.getElementById("text").value.trim();

        let language =
            document.getElementById("language").value;

        let consent =
            document.getElementById("consent").checked;

        let button =
            document.getElementById("generateBtn");

        let result =
            document.getElementById("result");

        if (!selectedFile) {

            showMessage(
                "Please upload a voice sample first.",
                "error"
            );

            return;
        }

        if (!consent) {

            showMessage(
                "Please confirm that you have permission to use this voice.",
                "error"
            );

            return;
        }

        if (!text) {

            showMessage(
                "Please enter some text.",
                "error"
            );

            return;
        }

        button.disabled = true;

        button.innerText =
            "â³ Generating... Please wait";

        result.style.display = "none";

        showMessage(
            "AI is generating your voice locally. The first generation may take longer because the model needs to load.",
            "loading"
        );

        try {

            let formData = new FormData();

            formData.append(
                "voice",
                selectedFile
            );

            formData.append(
                "text",
                text
            );

            formData.append(
                "language",
                language
            );


            let response = await fetch(
                "/api/clone",
                {
                    method: "POST",
                    body: formData
                }
            );


            let data = await response.json();


            if (!response.ok) {

                throw new Error(
                    data.error || "Generation failed."
                );

            }


            let audioPlayer =
                document.getElementById("audioPlayer");

            let download =
                document.getElementById("download");


            audioPlayer.src =
                data.audio_url +
                "?t=" +
                Date.now();


            download.href =
                data.audio_url;


            result.style.display =
                "block";


            showMessage(
                "Voice generated successfully.",
                "loading"
            );

        }

        catch (error) {

            showMessage(
                error.message,
                "error"
            );

        }

        finally {

            button.disabled = false;

            button.innerText =
                "âœ¨ Generate Voice";

        }

    }

from flask import Flask, request, jsonify, send_from_directory
from TTS.api import TTS

import os
import uuid
import shutil

app = Flask(__name__)

# --------------------------------------------------
# FOLDERS
# --------------------------------------------------

BASE_DIR = os.path.dirname(os.path.abspath(__file__))

VOICE_FOLDER = os.path.join(
    BASE_DIR,
    "voices"
)

OUTPUT_FOLDER = os.path.join(
    BASE_DIR,
    "outputs"
)

os.makedirs(VOICE_FOLDER, exist_ok=True)
os.makedirs(OUTPUT_FOLDER, exist_ok=True)


# --------------------------------------------------
# XTTS V2 MODEL
# --------------------------------------------------

MODEL_NAME = "tts_models/multilingual/multi-dataset/xtts_v2"

print("")
print("========================================")
print("      VoxClone AI - Starting")
print("========================================")
print("")

print("Loading XTTS-v2 model...")

try:

    import torch

    if torch.cuda.is_available():

        DEVICE = "cuda"

        print("NVIDIA GPU detected.")

    else:

        DEVICE = "cpu"

        print("Running on CPU.")

    tts = TTS(MODEL_NAME).to(DEVICE)

    print("XTTS-v2 loaded successfully.")

except Exception as error:

    print("")
    print("MODEL ERROR:")
    print(error)
    print("")
    tts = None


# --------------------------------------------------
# HOME PAGE
# --------------------------------------------------

@app.route("/")
def home():

    return send_from_directory(
        BASE_DIR,
        "index.html"
    )


# --------------------------------------------------
# GENERATED AUDIO
# --------------------------------------------------

@app.route("/outputs/<filename>")
def output_audio(filename):

    return send_from_directory(
        OUTPUT_FOLDER,
        filename
    )


# --------------------------------------------------
# VOICE CLONING API
# --------------------------------------------------

@app.route("/api/clone", methods=["POST"])
def clone_voice():

    if tts is None:

        return jsonify({
            "error":
            "The AI model could not load. Check the terminal for the error."
        }), 500


    # ----------------------------------------------
    # GET FORM DATA
    # ----------------------------------------------

    text = request.form.get(
        "text",
        ""
    ).strip()


    language = request.form.get(
        "language",
        "en"
    ).strip()


    voice_file = request.files.get(
        "voice"
    )


    # ----------------------------------------------
    # VALIDATION
    # ----------------------------------------------

    if not text:

        return jsonify({
            "error":
            "Please enter some text."
        }), 400


    if len(text) > 3000:

        return jsonify({
            "error":
            "Please keep the text under 3000 characters."
        }), 400


    if voice_file is None:

        return jsonify({
            "error":
            "Please upload a voice sample."
        }), 400


    # ----------------------------------------------
    # CHECK FILE
    # ----------------------------------------------

    original_name = voice_file.filename.lower()

    allowed_extensions = [
        ".wav",
        ".mp3",
        ".m4a"
    ]

    extension = os.path.splitext(
        original_name
    )[1]


    if extension not in allowed_extensions:

        return jsonify({
            "error":
            "Please upload WAV, MP3, or M4A audio."
        }), 400


    # ----------------------------------------------
    # SAVE REFERENCE AUDIO
    # ----------------------------------------------

    reference_name = (
        "reference_" +
        uuid.uuid4().hex +
        extension
    )

    reference_path = os.path.join(
        VOICE_FOLDER,
        reference_name
    )


    voice_file.save(
        reference_path
    )


    # ----------------------------------------------
    # OUTPUT FILE
    # ----------------------------------------------

    output_name = (
        "voice_" +
        uuid.uuid4().hex +
        ".wav"
    )

    output_path = os.path.join(
        OUTPUT_FOLDER,
        output_name
    )


    try:

        print("")
        print("----------------------------------------")
        print("Generating voice...")
        print("Language:", language)
        print("Reference:", reference_name)
        print("----------------------------------------")


        # ------------------------------------------
        # XTTS VOICE CLONING
        # ------------------------------------------

        tts.tts_to_file(

            text=text,

            speaker_wav=reference_path,

            language=language,

            file_path=output_path,

            split_sentences=True

        )


        print("Generation complete.")


        return jsonify({

            "success": True,

            "audio_url":
                "/outputs/" + output_name

        })


    except Exception as error:

        print("")
        print("GENERATION ERROR:")
        print(error)
        print("")


        return jsonify({

            "error":
                "Voice generation failed: " +
                str(error)

        }), 500


    finally:

        # ------------------------------------------
        # REMOVE TEMPORARY REFERENCE
        # ------------------------------------------

        try:

            if os.path.exists(reference_path):

                os.remove(
                    reference_path
                )

        except Exception:

            pass


# --------------------------------------------------
# START SERVER
# --------------------------------------------------

if __name__ == "__main__":

    print("")
    print("========================================")
    print("Website:")
    print("http://127.0.0.1:5000")
    print("========================================")
    print("")

    app.run(
        host="127.0.0.1",
        port=5000,
        debug=False
    )
